async function procesarMensajeBotWhatsApp(from, text) {
  if (!from) {
    return {
      status: 400,
      json: { error: 'falta from' }
    };
  }

  if (!text) {
    return {
      status: 400,
      json: { error: 'falta text' }
    };
  }

  const esTelefonoValido = !isNaN(from.trim()) && from.trim() !== '';
  const mensajeLimpio = text.toLowerCase().trim();

  if (!esTelefonoValido) {
    return {
      status: 400,
      json: { error: 'El telefono debe contener solo numeros' }
    };
  }

  const respuesta = await db.tx(async (t) => {
    let contacto = await t.oneOrNone(
      'SELECT * FROM contactos WHERE telefono = $1',
      [from]
    );

    let pasoActual = 1;

    if (!contacto) {
      contacto = await t.one(
        'INSERT INTO contactos(telefono) VALUES($1) RETURNING *',
        [from]
      );
    } else {
      const ultimoMensaje = await t.oneOrNone(
        'SELECT paso FROM mensajes WHERE contacto_id = $1 ORDER BY creado_en DESC LIMIT 1',
        [contacto.id]
      );

      if (ultimoMensaje) {
        pasoActual = ultimoMensaje.paso;
      }
    }

    let reply = '';
    let siguientePaso = pasoActual;
    let respuestaJson;

    if (mensajeLimpio === 'hola' && pasoActual !== 1) {
      siguientePaso = 2;
      reply = 'Hola, como te llamas?';

      await t.none(
        'INSERT INTO mensajes (contacto_id, direccion, texto, paso) VALUES ($1, $2, $3, $4)',
        [contacto.id, 'in', mensajeLimpio, siguientePaso]
      );

      await t.none(
        'INSERT INTO mensajes (contacto_id, direccion, texto, paso) VALUES ($1, $2, $3, $4)',
        [contacto.id, 'out', reply, siguientePaso]
      );

      return {
        status: 200,
        json: { reply }
      };
    }

    switch (pasoActual) {
      case 1:
        if (mensajeLimpio === 'hola') {
          siguientePaso = 2;
          reply = 'Hola, como te llamas?';

          await t.none(
            'INSERT INTO mensajes (contacto_id, direccion, texto, paso) VALUES ($1, $2, $3, $4)',
            [contacto.id, 'in', mensajeLimpio, siguientePaso]
          );
        } else {
          reply = "No te entendi. Por favor, escribe 'Hola' para comenzar.";
        }

        respuestaJson = { reply };
        break;

      case 2: {
        const categorias = await obtenerCategoriasDisponibles(t);

        siguientePaso = 3;
        reply = categorias.length > 0
          ? `Bienvenido ${text}. Primero elige una categoria:\n\n${formatearCategorias(categorias)}`
          : `Bienvenido ${text}, por ahora no tenemos categorias con productos disponibles.`;

        await t.none(
          'UPDATE contactos SET nombre = $1 WHERE id = $2',
          [text, contacto.id]
        );

        await t.none(
          'INSERT INTO mensajes (contacto_id, direccion, texto, paso) VALUES ($1, $2, $3, $4)',
          [contacto.id, 'in', text, siguientePaso]
        );

        respuestaJson = {
          reply,
          categorias
        };
        break;
      }

      case 3: {
        const categoriaSeleccionada = await obtenerCategoriaPorNombre(t, mensajeLimpio);

        if (!categoriaSeleccionada) {
          const categorias = await obtenerCategoriasDisponibles(t);

          reply = categorias.length > 0
            ? `No encontre esa categoria. Elige una de estas opciones:\n\n${formatearCategorias(categorias)}`
            : 'Lo siento, actualmente no tenemos categorias con productos disponibles.';

          respuestaJson = { reply };
          break;
        }

        const productos = await obtenerProductosPorCategoria(t, categoriaSeleccionada.id);

        siguientePaso = 4;
        reply = productos.length > 0
          ? `Estos son los productos de ${categoriaSeleccionada.nombre}:\n\nResponde con el nombre del producto que deseas ordenar.`
          : `Por ahora no tenemos productos disponibles en ${categoriaSeleccionada.nombre}.`;

        await t.none(
          'INSERT INTO mensajes (contacto_id, direccion, texto, paso) VALUES ($1, $2, $3, $4)',
          [contacto.id, 'in', categoriaSeleccionada.nombre, siguientePaso]
        );

        respuestaJson = {
          reply,
          productos
        };
        break;
      }

      case 4: {
        const ultimaCategoria = await t.oneOrNone(
          `SELECT texto
          FROM mensajes
          WHERE contacto_id = $1 AND direccion = 'in' AND paso = 4
          ORDER BY creado_en DESC
          LIMIT 1`,
          [contacto.id]
        );

        const productoSeleccionado = await t.oneOrNone(
          `SELECT
            p.id,
            p.nombre,
            p.descripcion,
            p.imagen_url,
            p.stock,
            c.nombre AS categoria
          FROM productos p
          JOIN categorias c
          ON c.id = p.categoria_id
          WHERE p.activo = true
          AND LOWER(p.nombre) = $1
          AND ($2::text IS NULL OR LOWER(c.nombre) = LOWER($2::text))
          LIMIT 1`,
          [mensajeLimpio, ultimaCategoria?.texto ?? null]
        );

        if (!productoSeleccionado) {
          const productos = ultimaCategoria
            ? await obtenerProductosPorCategoriaNombre(t, ultimaCategoria.texto)
            : await obtenerProductosActivos(t);
          const nombresProductos = productos.map((producto) => producto.nombre).join(', ');

          reply = nombresProductos
            ? `Lo siento, no encontre ese producto. Actualmente tenemos disponible: ${nombresProductos}.`
            : 'Lo siento, actualmente no tenemos productos disponibles.';

          respuestaJson = { reply };
          break;
        }

        if (productoSeleccionado.stock <= 0) {
          reply = `Lo siento, ${productoSeleccionado.nombre} ya no tiene stock disponible.`;
          respuestaJson = { reply };
          break;
        }

        const stockActualizado = await t.oneOrNone(
          `UPDATE productos
          SET stock = stock - 1
          WHERE id = $1 AND stock > 0
          RETURNING stock`,
          [productoSeleccionado.id]
        );

        if (!stockActualizado) {
          reply = `Lo siento, ${productoSeleccionado.nombre} ya no tiene stock disponible.`;
          respuestaJson = { reply };
          break;
        }

        const totalSolicitudes = await t.one('SELECT count(*) FROM solicitudes');
        const numeroFormateado = String(Number(totalSolicitudes.count) + 1).padStart(3, '0');
        const evento_id = `ev-${numeroFormateado}`;

        await t.none(
          'INSERT INTO mensajes (contacto_id, direccion, texto, paso, imagen_url) VALUES ($1, $2, $3, $4, $5)',
          [
            contacto.id,
            'in',
            productoSeleccionado.nombre,
            siguientePaso,
            productoSeleccionado.imagen_url
          ]
        );

        await t.none(
          'INSERT INTO solicitudes (producto_id, estado, evento_id, contacto_id) VALUES($1, $2, $3, $4)',
          [
            productoSeleccionado.id,
            'pendiente',
            evento_id,
            contacto.id
          ]
        );

        reply = `Tu pedido de ${productoSeleccionado.nombre} esta en estado pendiente.`;
        respuestaJson = {
          reply,
          producto: {
            nombre: productoSeleccionado.nombre,
            descripcion: productoSeleccionado.descripcion,
            categoria: productoSeleccionado.categoria,
            stock: stockActualizado.stock,
            img: productoSeleccionado.imagen_url
          }
        };
        break;
      }

      default:
        reply = "Algo salio mal con tu sesion. Por favor escribe 'Hola' para comenzar.";
        respuestaJson = { reply };
        break;
    }

    if (reply) {
      await t.none(
        'INSERT INTO mensajes (contacto_id, direccion, texto, paso) VALUES ($1, $2, $3, $4)',
        [contacto.id, 'out', reply, siguientePaso]
      );
    }

    return {
      status: 200,
      json: respuestaJson
    };
  });

  return respuesta;
}

async function obtenerProductosActivos(t) {
  return t.any(`
    SELECT
      p.id,
      p.nombre,
      p.descripcion,
      p.imagen_url AS img,
      p.stock,
      c.nombre AS categoria
    FROM productos p
    JOIN categorias c
    ON c.id = p.categoria_id
    WHERE p.activo = true
    ORDER BY c.nombre, p.nombre
  `);
}

async function obtenerCategoriasDisponibles(t) {
  return t.any(`
    SELECT
      c.id,
      c.nombre,
      c.descripcion
    FROM categorias c
    JOIN productos p
    ON p.categoria_id = c.id
    WHERE p.activo = true
    GROUP BY c.id, c.nombre, c.descripcion
    ORDER BY c.nombre
  `);
}

async function obtenerCategoriaPorNombre(t, nombreCategoria) {
  return t.oneOrNone(
    `SELECT id, nombre, descripcion
    FROM categorias
    WHERE LOWER(nombre) = $1
    LIMIT 1`,
    [nombreCategoria]
  );
}

async function obtenerProductosPorCategoria(t, categoriaId) {
  return t.any(
    `SELECT
      p.id,
      p.nombre,
      p.descripcion,
      p.imagen_url AS img,
      p.stock,
      c.nombre AS categoria
    FROM productos p
    JOIN categorias c
    ON c.id = p.categoria_id
    WHERE p.activo = true AND p.categoria_id = $1
    ORDER BY p.nombre`,
    [categoriaId]
  );
}

async function obtenerProductosPorCategoriaNombre(t, nombreCategoria) {
  return t.any(
    `SELECT
      p.id,
      p.nombre,
      p.descripcion,
      p.imagen_url AS img,
      p.stock,
      c.nombre AS categoria
    FROM productos p
    JOIN categorias c
    ON c.id = p.categoria_id
    WHERE p.activo = true AND LOWER(c.nombre) = LOWER($1)
    ORDER BY p.nombre`,
    [nombreCategoria]
  );
}

function formatearCategorias(categorias) {
  return categorias
    .map((categoria) => {
      const descripcion = categoria.descripcion ? ` - ${categoria.descripcion}` : '';
      return `- ${categoria.nombre}${descripcion}`;
    })
    .join('\n');
}

function formatearProducto(producto) {
  const descripcion = producto.descripcion || 'Sin descripcion';
  const stock = producto.stock > 0 ? `${producto.stock} disponible(s)` : 'Sin stock';

  return [
    producto.nombre,
    `Categoria: ${producto.categoria}`,
    `Descripcion: ${descripcion}`,
    `Stock: ${stock}`
  ].join('\n');
}

  const { evento_id, solicitud_id } = req.body;

  }

  try {
    const respuesta = await db.tx(async (t) => {
      const solicitudConfirmada = await t.oneOrNone(`
        UPDATE solicitudes s
        SET estado = 'confirmada', evento_id = $1

        WHERE s.id = $2
        AND s.estado = 'pendiente'
        AND c.id = s.contacto_id
        AND p.id = s.producto_id
        RETURNING
          s.id,
          s.estado,
          s.evento_id,
          c.telefono,
          c.nombre AS contacto_nombre,
          p.nombre AS producto_nombre
      `, [evento_id, solicitud_id]);

      if (!solicitudConfirmada) {
        const solicitudExiste = await t.oneOrNone(
          `SELECT
            s.id,
            s.estado,
            c.telefono,
            p.nombre AS producto_nombre
          FROM solicitudes s
          LEFT JOIN contactos c
          ON c.id = s.contacto_id
          LEFT JOIN productos p
          ON p.id = s.producto_id
          WHERE s.id = $1`,
          [solicitud_id]
        );
        
        if (!solicitudExiste) {
          return { status: 404, json: { error: 'Solicitud no encontrada' } };
        }
        
        return {
          status: 200,
          json: { mensaje: 'La solicitud ya estaba confirmada o no disponible' },
          notificarCliente: false
        };
      }

      return {
        status: 200,
        json: { mensaje: 'Solicitud confirmada exitosamente' },
        notificarCliente: true,
        solicitud: solicitudConfirmada
      };
    });

    if (respuesta.notificarCliente && respuesta.solicitud?.telefono) {
      const mensajeConfirmacion = `Tu pedido de ${respuesta.solicitud.producto_nombre} fue confirmado exitosamente.`;

      try {
        await enviarMensajeWhatsApp(respuesta.solicitud.telefono, mensajeConfirmacion);
      } catch (error) {
        logErrorWhatsApp(error);
      }
    }

    return res.status(respuesta.status).json(respuesta.json);

  } catch (error) {
    console.error('Error procesando confirmación de webhook:', error);
    return res.status(500).json({ error: 'Error interno de la base de datos' });
  }
});

async function enviarMensajeWhatsApp(to, mensaje) {
  const url = `https://graph.facebook.com/v25.0/${process.env.WHATSAPP_PHONE_NUMBER_ID}/messages`

 const response = await axios.post(
    url,
    {
      messaging_product: 'whatsapp',
      recipient_type: 'individual',
      to,
      type: 'text',
      text: {
        preview_url: false,
        body: mensaje
      }
    },
    {
      headers: {
        Authorization: `Bearer ${process.env.WHATSAPP_TOKEN}`,
        'Content-Type': 'application/json'
      }
    }
  );


  return response.data;
}

async function enviarImagenWhatsApp(to, imageUrl, caption = '') {
  const url = `https://graph.facebook.com/v25.0/${process.env.WHATSAPP_PHONE_NUMBER_ID}/messages`

  const response = await axios.post(
    url,
    {
      messaging_product: 'whatsapp',
      recipient_type: 'individual',
      to,
      type: 'image',
      image: {
        link: imageUrl,
        caption
      }
    },
    {
      headers: {
        Authorization: `Bearer ${process.env.WHATSAPP_TOKEN}`,
        'Content-Type': 'application/json'
      }
    }
  );

  return response.data;
}

function logErrorWhatsApp(error) {
  console.error('Error enviando mensaje por WhatsApp:', {
    status: error.response?.status,
    data: error.response?.data || error.message
  });
}

  console.log('Verificando webhook:', req.query);

  if (mode === 'subscribe' && token === process.env.WHATSAPP_VERIFY_TOKEN) {

  }

});

app.post('/webhook/whatsapp', async (req, res) => {
  try {
    const value = req.body.entry?.[0]?.changes?.[0]?.value;
    const message = value?.messages?.[0];

    if (!message) {
      return res.sendStatus(200);
    }

    if (message.type !== 'text') {
      return res.sendStatus(200);
    }

    const from = message.from;
    const text = message.text?.body;

    console.log('Mensaje recibido desde WhatsApp:', {
      from,
      text
    });

    const respuesta = await procesarMensajeBotWhatsApp(from, text);

    try {
      if (respuesta.status === 200 && respuesta.json.reply) {
        await enviarMensajeWhatsApp(from, respuesta.json.reply);
      }

      if (respuesta.status === 200 && respuesta.json.productos?.length > 0) {
        for (const producto of respuesta.json.productos) {
          if (!producto.img) continue;

          await enviarImagenWhatsApp(from, producto.img, formatearProducto(producto));
        }
      }
    } catch (error) {
      logErrorWhatsApp(error);
    }

    return res.sendStatus(200);

  } catch (error) {
    console.error('Error en webhook WhatsApp:', error);
    return res.sendStatus(500);
  }
});
